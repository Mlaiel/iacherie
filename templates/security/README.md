### Installation

```bash
# Install required dependencies
pip install numpy pillow opencv-python librosa soundfile scikit-learn nltk spacy pydantic cryptography

# Clone the repository
git clone https://github.com/Mlaiel/IA Chérie.git
cd IA Chérie/templates/security
```

### Basic Usage

```python
from templates.security import (
    ContentWatermarkingTemplate,
    DigitalRightsManagementTemplate,
    AnomalyDetectionTemplate,
    FraudDetectionTemplate,
    get_security_registry
)

# Initialize content watermarking
watermark_config = WatermarkConfig(
    content_id="creator_content_001",
    creator_id="creator_123",
    content_type=ContentType.IMAGE,
    watermark_type=WatermarkType.INVISIBLE,
    strength=WatermarkStrength.HIGH
)

watermarking = ContentWatermarkingTemplate()
watermarked_content, metadata = watermarking.add_watermark(
    content=image_data,
    config=watermark_config
)

# Initialize fraud detection
fraud_config = FraudConfig(
    detection_id="fraud_detector_001",
    creator_id="creator_123",
    fraud_types={FraudType.PAYMENT_FRAUD, FraudType.CONTENT_THEFT},
    sensitivity_level="high"
)

fraud_detector = FraudDetectionTemplate()
detection_id = fraud_detector.register_detector(fraud_config)

# Detect fraud in transaction
fraud_event = fraud_detector.detect_fraud(
    detection_id=detection_id,
    transaction_data=transaction_data
)

# Get security registry
registry = get_security_registry()
available_templates = registry.list_templates()
```

### Advanced Configuration

```python
# Configure DRM for content protection
drm_config = DRMConfig(
    content_id="premium_content_001",
    creator_id="creator_123",
    license_type=LicenseType.EXCLUSIVE,
    protection_level=ProtectionLevel.MAXIMUM,
    usage_rights={UsageRight.VIEW, UsageRight.DOWNLOAD},
    blockchain_anchored=True
)

drm = DigitalRightsManagementTemplate()
protected_content, drm_metadata = drm.protect_content(content, drm_config)

# Issue license to user
license_data = drm.issue_license(
    drm_config=drm_config,
    recipient="user_456",
    custom_terms={"expiration_date": datetime.utcnow() + timedelta(days=30)}
)
```

## 🔗 Integration Examples

### Flask Integration

```python
from flask import Flask, request
from templates.security import SecurityTemplateManager

app = Flask(__name__)
security_manager = SecurityTemplateManager()

@app.before_request
def security_check():
    # Fraud detection
    fraud_detector = security_manager.get_template('fraud_detection')
    fraud_event = fraud_detector.detect_fraud(
        detection_id="web_transactions",
        transaction_data={
            'ip_address': request.remote_addr,
            'user_agent': request.user_agent.string,
            'user_id': get_current_user_id(),
            'timestamp': datetime.utcnow()
        }
    )
    
    if fraud_event.blocked:
        abort(403, "Transaction blocked due to fraud detection")
```

### FastAPI Integration

```python
from fastapi import FastAPI, Depends, HTTPException
from templates.security import get_security_registry

app = FastAPI()
registry = get_security_registry()

async def validate_content_access(content_id: str, user_id: str):
    drm_template = registry.get_template('digital_rights_management')
    
    access_granted, details = drm_template.validate_access(
        content_id=content_id,
        user_id=user_id,
        requested_rights={UsageRight.VIEW}
    )
    
    if not access_granted:
        raise HTTPException(status_code=403, detail=details['error'])
    
    return details

@app.get("/content/{content_id}")
async def get_content(
    content_id: str,
    access_details = Depends(validate_content_access)
):
    # Return protected content
    return {"content": "protected_content", "access": access_details}
```

### Django Integration

```python
from django.middleware.common import CommonMiddleware
from templates.security import AnomalyDetectionTemplate

class SecurityMiddleware(CommonMiddleware):
    def __init__(self, get_response):
        self.get_response = get_response
        self.anomaly_detector = AnomalyDetectionTemplate()
        
    def __call__(self, request):
        # Anomaly detection
        user_data = {
            'user_id': str(request.user.id) if request.user.is_authenticated else None,
            'ip_address': self.get_client_ip(request),
            'user_agent': request.META.get('HTTP_USER_AGENT', ''),
            'request_path': request.path,
            'request_method': request.method
        }
        
        anomalies = self.anomaly_detector.detect_anomalies(
            detection_id="web_requests",
            data=[user_data]
        )
        
        if any(a.anomaly_level == AnomalyLevel.CRITICAL for a in anomalies):
            return HttpResponseForbidden("Request blocked due to anomaly detection")
        
        response = self.get_response(request)
        return response
```
