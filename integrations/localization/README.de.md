## 🚀 Schnellstart

### Installation

```bash
# Repository klonen (nur autorisierte Benutzer)
git clone https://github.com/Mlaiel/Ainflue.git
cd Ainflue/integrations/localization

# Abhängigkeiten installieren
pip install -r requirements.txt

# Umgebung konfigurieren
export AINFLUE_LOCALIZATION_API_KEY="your-api-key"
export REDIS_URL="redis://localhost:6379/0"
```

### Grundlegende Verwendung

```python
from integrations.localization import get_localization_manager

# Lokalisierungs-Manager initialisieren
localization = get_localization_manager()

# Grundlegende Übersetzung
result = await localization['translation'].translate(
    content="Hallo, willkommen bei Ainflue!",
    source_language="de",
    target_language="en",
    domain="social_media"
)

# Kulturelle Anpassung
adapted = await localization['cultural'].adapt_content(
    content=result.translated_text,
    target_culture="american_casual",
    context="business"
)

# Echtzeit-Lokalisierung
real_time = localization['real_time']
response = await real_time.process_realtime_request(
    content="Live-Streaming-Inhalt...",
    target_language="fr",
    mode="streaming"
)
```

### Konfigurationsdatei

```yaml

localization:
  default_locale: "de"
  supported_locales: ["de", "en", "fr", "ar", "es", "pt", "ja", "ko", "zh"]
  
  translation_service:
    provider: "neural_ai"
    fallback_provider: "google_translate"

  cultural_adaptation:
    enabled: true
    sensitivity_level: "high"
    context_awareness: true
    
  real_time:
    streaming_enabled: true
    websocket_port: 8765
    max_concurrent_streams: 1000
    
  analytics:
    enabled: true
    metrics_retention_days: 365
    performance_monitoring: true
```

## 📈 API-Referenz

### Kern-Lokalisierungs-Manager

```python
# Lokalisierungs-Manager abrufen
manager = get_localization_manager()

# Verfügbare Komponenten
components = {
    'i18n': InternationalizationManager(),
    'translation': AITranslationEngine(), 
    'cultural': CulturalAdaptationEngine(),
    'regional': RegionalComplianceManager(),
    'content': ContentLocalizationProcessor(),
    'voice': VoiceLocalizationEngine(),
    'analytics': LocalizationAnalytics(),
    'real_time': RealtimeLocalizationEngine()
}
```

### Übersetzungs-Engine

```python
# KI-Übersetzung
result = await translation_engine.translate(
    content="Ihr Inhalt hier",
    source_language="de",
    target_language="en",
    domain="social_media",  # oder "business", "technical", "creative"
    quality_level="high",   # oder "medium", "fast"
    cultural_context={
        "formality": "formal",
        "audience": "business",
        "region": "germany"
    }
)
```

### Echtzeit-Verarbeitung

```python
# Echtzeit-Lokalisierung
request = RealtimeLocalizationRequest(
    content="Live-Content-Stream...",
    source_language="de",
    target_language="en",
    real_time_mode="streaming",  # oder "instant", "batch"
    priority="high"
)

response = await real_time_engine.process_realtime_request(request)
```
