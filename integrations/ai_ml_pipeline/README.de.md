### Quick-Start-Beispiel
```python
import asyncio
from integrations.ai_ml_pipeline import create_complete_ai_ml_pipeline

async def main():
    # Konfiguration laden
    config = {
        'database_url': 'postgresql://localhost/ainflue',
        'redis_url': 'redis://localhost:6379',
        'gpu_enabled': True,
        'monitoring_enabled': True
    }
    
    # Pipeline initialisieren
    pipeline = await create_complete_ai_ml_pipeline(config)
    
    # KI/ML-Operationen starten
    print("✅ KI/ML-Pipeline erfolgreich initialisiert")
    print(f"📊 Komponenten: {len(pipeline)} aktiv")
    
    # Beispiel: Modell registrieren und deployen
    model_registry = pipeline['model_registry']
    deployment_manager = pipeline['deployment_manager']
    
    # Modell registrieren
    model_id = await model_registry.register_model(
        name="creator_content_optimizer_v2",
        version="2.1.0",
        model_path="s3://models/content_optimizer_v2.pt"
    )
    
    # In Produktion deployen
    deployment_id = await deployment_manager.deploy_model(
        model_id=model_id,
        environment="production",
        strategy="blue_green"
    )
    
    print(f"🚀 Modell deployed: {deployment_id}")

if __name__ == "__main__":
    asyncio.run(main())
```
