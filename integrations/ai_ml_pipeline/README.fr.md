### Exemple Démarrage Rapide
```python
import asyncio
from integrations.ai_ml_pipeline import create_complete_ai_ml_pipeline

async def main():
    # Charger configuration
    config = {
        'database_url': 'postgresql://localhost/iacherie',
        'redis_url': 'redis://localhost:6379',
        'gpu_enabled': True,
        'monitoring_enabled': True
    }
    
    # Initialiser pipeline
    pipeline = await create_complete_ai_ml_pipeline(config)
    
    # Démarrer opérations IA/ML
    print("✅ Pipeline IA/ML initialisé avec succès")
    print(f"📊 Composants : {len(pipeline)} actifs")
    
    # Exemple : Enregistrer et déployer modèle
    model_registry = pipeline['model_registry']
    deployment_manager = pipeline['deployment_manager']
    
    # Enregistrer modèle
    model_id = await model_registry.register_model(
        name="creator_content_optimizer_v2",
        version="2.1.0",
        model_path="s3://models/content_optimizer_v2.pt"
    )
    
    # Déployer en production
    deployment_id = await deployment_manager.deploy_model(
        model_id=model_id,
        environment="production",
        strategy="blue_green"
    )
    
    print(f"🚀 Modèle déployé : {deployment_id}")

if __name__ == "__main__":
    asyncio.run(main())
```
