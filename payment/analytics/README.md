## 🚀 GETTING STARTED

```python
from payment.analytics import (
    PaymentGatewayAnalytics,
    RevenueTracker,
    FraudDetectionAnalytics,
    FinancialReporting,
    CreatorEarningsAnalytics,
    PaymentPerformanceMonitor,
    RealTimeDashboard
)

# Initialize analytics engine
config = {
    "commission_rates": {"default": 0.15},
    "risk_thresholds": {"high": 0.8},
    "accounting_standards": "GAAP"
}

# Revenue tracking
revenue_tracker = RevenueTracker(config)
await revenue_tracker.initialize()
revenue_metrics = await revenue_tracker.track_platform_revenue(
    period_start=datetime(2025, 1, 1),
    period_end=datetime(2025, 1, 31)
)

# Fraud detection
fraud_detector = FraudDetectionAnalytics(config)
await fraud_detector.initialize()
fraud_alert = await fraud_detector.analyze_transaction_fraud(
    transaction={"amount": 1000, "currency": "USD"},
    user_context={"user_id": "user_123", "ip_address": "192.168.1.1"}
)

# Creator earnings
creator_analytics = CreatorEarningsAnalytics(config)
creator_earnings = await creator_analytics.analyze_creator_earnings(
    creator_id="creator_456",
    period_start=datetime(2025, 1, 1),
    period_end=datetime(2025, 1, 31)
)
```
