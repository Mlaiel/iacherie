### Quick Start Example

```python
import requests

# Authentication
headers = {
    "Authorization": "Bearer YOUR_JWT_TOKEN",
    "Content-Type": "application/json"
}

# Create a collaboration request
collaboration_data = {
    "requester_id": "creator_123",
    "collaboration_type": "music_production",
    "project_title": "Summer Electronic EP",
    "project_description": "Collaborative electronic music project",
    "required_skills": ["music_production", "mixing", "mastering"],
    "preferred_genres": ["electronic", "house", "techno"],
    "target_audience": {"age_range": "18-35", "interests": ["electronic_music"]},
    "budget_range": {"min": 1000, "max": 5000},
    "timeline": {
        "start_date": "2025-02-01T00:00:00Z",
        "end_date": "2025-04-01T00:00:00Z"
    },
    "revenue_share_model": "performance_based"
}

# Find compatible creators
response = requests.post(
    "https://api.ainflue.com/api/v1/collaboration/matching/find-creators",
    headers=headers,
    json={
        "creator_id": "creator_123",
        "collaboration_type": "music_production",
        "matching_criteria": ["genre_compatibility", "audience_overlap"],
        "max_results": 10
    }
)

matches = response.json()
print(f"Found {len(matches['data']['matches'])} compatible creators")
```

### Content Distribution Example

```python
# Distribute content across multiple platforms
distribution_request = {
    "content_id": "content_456",
    "content_format": "audio_track",
    "target_platforms": ["spotify", "apple_music", "youtube", "soundcloud"],
    "synchronization_mode": "immediate",
    "metadata_overrides": {
        "spotify": {"genre": "Electronic", "mood": "Energetic"},
        "youtube": {"title": "Epic Electronic Track - Official Audio"}
    }
}

response = requests.post(
    "https://api.ainflue.com/api/v1/distribution/content/distribute",
    headers=headers,
    json=distribution_request
)

distribution_result = response.json()
print(f"Distribution initiated to {len(distribution_request['target_platforms'])} platforms")
```
