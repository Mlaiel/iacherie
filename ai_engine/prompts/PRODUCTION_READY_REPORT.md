```python
from ai.prompts import prompt_manager

# Générer un prompt pour musicien
music_prompt = prompt_manager.generate_content_creator_prompt(
    creator_type="musician",
    content_format="audio",
    category="creation",
    user_preferences={
        "genre": "electronic",
        "tempo": "128",
        "mood": "energetic"
    }
```
