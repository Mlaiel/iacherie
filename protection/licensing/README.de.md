
## 📖 Verwendungsbeispiele

### Grundlegende Lizenzgenerierung

```python
from backend.content_protection.licensing import LicensingSystem

# Lizenzierungssystem initialisieren
licensing = LicensingSystem({
    'database_url': 'postgresql://user:pass@localhost/licensing',
    'ai_models_path': './models',
    'blockchain_network': 'ethereum',
    'legal_templates_path': './templates'
})

# Automatisierte Lizenz generieren
license_request = {
    'content_type': 'musical_work',
    'territory': ['US', 'EU', 'UK'],
    'usage_rights': ['streaming', 'download', 'public_performance'],
    'revenue_share': 70.0,
    'exclusivity': False,
    'duration': {'years': 5}
}

result = await licensing.generate_automated_license(
    content_id='track_12345',
    parameters=license_request
)

print(f"Lizenz generiert: {result['license_id']}")
```

### KI-Vertragsgenerierung

```python
# KI-gestützten Vertrag generieren
contract_params = {
    'contract_type': 'music_licensing',
    'jurisdiction': 'eu_general',
    'parties': {
        'licensor': {'name': 'Künstlername', 'country': 'Deutschland'},
        'licensee': {'name': 'Plattform GmbH', 'country': 'Deutschland'}
    },
    'financial_terms': {
        'royalty_percentage': 75.0,
        'minimum_guarantee': 10000.0,
        'currency': 'EUR'
    },
    'content_details': {
        'title': 'Songtitel',
        'description': 'Originale Musikkomposition'
    }
}

contract_result = await licensing.generate_ai_contract(
    contract_parameters=contract_params,
    custom_clauses=['Höhere Gewalt Klausel', 'Schiedsgerichtsklausel']
)

print(f"Vertrag generiert: {contract_result['contract_id']}")
```
