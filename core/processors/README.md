## 🚀 USAGE EXAMPLES

### 🎵 Audio Processing
```python
from backend.core.processors import AudioProcessor, AudioProcessingConfig

# Initialize audio processor
config = AudioProcessingConfig(
    enable_ai_enhancement=True,
    target_format="mp3",
    quality="high",
    enable_fingerprinting=True
)

processor = await AudioProcessor.create(
    db_session=db_session,
    redis_client=redis_client,
    config=config
)

# Process audio file
result = await processor.process_audio(
    audio_data=audio_bytes,
    options={
        "enhance_quality": True,
        "normalize_volume": True,
        "remove_noise": True
    }
)
```

### 🎬 Video Processing
```python
from backend.core.processors import VideoProcessor, VideoProcessingConfig

# Initialize video processor
config = VideoProcessingConfig(
    enable_ai_enhancement=True,
    target_resolution="1080p",
    enable_scene_detection=True,
    enable_subtitles=True
)

processor = await VideoProcessor.create(
    db_session=db_session,
    redis_client=redis_client,
    config=config
)

# Process video file

    video_data=video_bytes,
    options={

        "generate_thumbnails": True,
        "extract_scenes": True,
        "create_preview": True
    }

```

### 📦 Batch Processing
```python
from backend.core.processors import BatchProcessor, BatchProcessingConfig

# Initialize batch processor
config = BatchProcessingConfig(
    max_concurrent_jobs=50,
    enable_progress_tracking=True,
    enable_error_recovery=True
)

processor = await BatchProcessor.create(
    db_session=db_session,
    redis_client=redis_client,
    config=config
)

# Submit batch job
job_id = await processor.submit_batch_job(
    files=file_list,
    processor_type="audio",
    processing_options=options
)

# Monitor progress
progress = await processor.get_job_progress(job_id)
```

## 🚀 Integration Examples

### FastAPI Integration

```python
from fastapi import FastAPI, UploadFile
from IA_Influencer_Agent.backend.core.processors import create_processor_manager

app = FastAPI()
processor_manager = None

@app.on_event("startup")
async def startup():
    global processor_manager
    processor_manager = await create_processor_manager(
        db_session, redis_client, auto_initialize=True
    )

@app.post("/process/audio")
async def process_audio(file: UploadFile):
    audio_processor = processor_manager.get_processor("audio")
    result = await audio_processor.process_upload(file)
    return result

@app.post("/process/video") 
async def process_video(file: UploadFile):
    video_processor = processor_manager.get_processor("video")
    result = await video_processor.process_upload(file)
    return result
```

### Celery Integration

```python
from celery import Celery
from IA_Influencer_Agent.backend.core.processors import get_processor_by_type

celery_app = Celery('processors')

    return result
```
