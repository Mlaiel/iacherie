# Lizenzierungssystem initialisieren
licensing = LicensingEngine()
await licensing.initialize()

# Content-Lizenzierung verarbeiten
license_result = await licensing.create_license(
    content_id="content_123",
    creator_id="user_456", 
    license_type="commercial",
    territory="worldwide",
    duration_years=5

royalty_processor = RoyaltyProcessor()
