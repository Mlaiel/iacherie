from backend.core.licensing import LicensingEngine, RoyaltyProcessor

licensing = LicensingEngine()

# Process content licensing
license_result = await licensing.create_license(
    content_id="content_123",
    creator_id="user_456", 
    license_type="commercial",
    territory="worldwide",
    duration_years=5
)

# Calculate and distribute royalties
royalty_processor = RoyaltyProcessor()
