## 📝 Usage Examples / Exemples d'Utilisation

### Vector Store Manager
```python
from backend.database.vector_stores import VectorStoreManager

# Initialize manager
manager = VectorStoreManager()

# Store vectors
await manager.store_vectors(
    content_type="audio",
    vectors=[(content_id, embedding) for content_id, embedding in audio_embeddings]
)

# Search similar content
results = await manager.search_similar(
    content_type="audio",
    query_vector=query_embedding,
    limit=10,
    threshold=0.8
)
```

### Similarity Search
```python
from backend.database.vector_stores import SimilaritySearchEngine

# Initialize search engine
search_engine = SimilaritySearchEngine()

# Perform similarity search
search_results = await search_engine.search_similar_content(
    content_type="image",
    query_vector=image_embedding,
    search_config=SimilaritySearchConfig(
        algorithm=SimilarityAlgorithm.COSINE,
        limit=20,
        threshold=0.85
    )
)
```

### Vector Clustering
```python
from backend.database.vector_stores import VectorClusteringEngine

# Initialize clustering engine
clustering = VectorClusteringEngine()

# Perform clustering analysis
cluster_results = await clustering.cluster_vectors(
    content_type="text",
    config=ClusteringConfig(
        algorithm=ClusteringAlgorithm.DBSCAN,
        eps=0.3,
        min_samples=5
    )
)

# Detect duplicates
duplicates = await clustering.detect_duplicates(
    content_type="text",
    similarity_threshold=0.95
)
```

### Embedding Generation
```python
from backend.database.vector_stores import EmbeddingGenerator

# Initialize generator
generator = EmbeddingGenerator()

# Generate embeddings for different content types
text_embeddings = await generator.generate_text_embeddings(
    texts=["Sample text content"],
    model_name="sentence-transformers/all-MiniLM-L6-v2"
)

image_embeddings = await generator.generate_image_embeddings(
    image_paths=["path/to/image.jpg"],
    model_name="openai/clip-vit-base-patch32"
)

audio_embeddings = await generator.generate_audio_embeddings(
    audio_paths=["path/to/audio.wav"],
    model_name="facebook/wav2vec2-base-960h"
)
```
