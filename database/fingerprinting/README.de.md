repo = FingerprintRepository(db_manager, encryption_manager)

    content_id="content_123",
    fingerprint_data={
        "primary_hash": "abc123...",
        "feature_vector": [0.1, 0.2, ...],
        "confidence_score": 0.95

    user_id="user_456"
