## 🔧 **Exemples d'Utilisation**

### **Exemple Workflow Créateur Complet**
```python
from examples import ExampleOrchestrator, CreatorWorkflowManager
from examples import AnalyticsExample, MonetizationExample

# Initialiser système exemples ultra-avancé
orchestrator = ExampleOrchestrator()
workflow_manager = CreatorWorkflowManager()
analytics = AnalyticsExample()
monetization = MonetizationExample()

async def run_comprehensive_creator_workflow():
    """Exemple workflow créateur complet présentant toutes les fonctionnalités"""
    
    # 1. Créer Profil Créateur
    creator_profile = await workflow_manager.create_creator_profile({
        'creator_id': 'creator_001',
        'creator_type': 'musician',
        'name': 'Alex Rodriguez',
        'specialization': 'Electronic Music Producer',
        'experience_level': 'advanced',
        'target_audience': {
            'age_range': '18-35',
            'locations': ['US', 'Europe', 'Canada'],
            'interests': ['electronic_music', 'festivals', 'technology']
        },
        'content_preferences': {
            'formats': ['audio', 'video', 'live_streams'],
            'platforms': ['spotify', 'youtube', 'twitch', 'instagram'],
            'frequency': 'weekly'
        },
        'monetization_goals': {
            'primary': 'streaming_revenue',
            'secondary': ['merchandise', 'live_performances', 'courses'],
            'target_monthly_revenue': 10000
        }
    })
    
    # 2. Générer Exemples Personnalisés
    personalized_examples = await workflow_manager.generate_personalized_examples(
        creator_profile.creator_id
    )
    
    # 3. Enregistrer Exemples dans Orchestrateur
    for example_config in personalized_examples:
        # Créer instance exemple basée sur config
        if example_config.example_id.startswith('audio_processing'):
            from examples.ai_processing_pipeline_examples import AudioProcessingExample
            example = AudioProcessingExample(example_config)
        elif example_config.example_id.startswith('music_distribution'):
            from examples.platform_integration_example import MusicDistributionExample
            example = MusicDistributionExample(example_config)
        
        await orchestrator.register_example(example)
    
    # 4. Exécuter Suite Exemples Complète
    suite_results = await orchestrator.execute_comprehensive_suite(
        creator_type='musician',
        complexity_filter='advanced'
    )
    
    # 5. Analyser Résultats et Générer Insights
    analytics_results = await analytics.analyze_creator_performance({
        'creator_id': creator_profile.creator_id,
        'execution_results': suite_results,
        'time_period': '30_days',
        'metrics': ['engagement', 'revenue', 'growth', 'quality']
    })
    
    # 6. Générer Recommandations Monétisation
    monetization_insights = await monetization.generate_revenue_optimization({
        'creator_profile': creator_profile,
        'performance_data': analytics_results,
        'market_trends': True,
        'competitor_analysis': True
    })
    
    return {
        'creator_profile': creator_profile,
        'personalized_examples': len(personalized_examples),
        'suite_results': suite_results,
        'analytics_results': analytics_results,
        'monetization_insights': monetization_insights
    }

# Exécuter workflow complet
workflow_results = await run_comprehensive_creator_workflow()
print(f"Workflow créateur terminé : {workflow_results['suite_results']['successful_examples']} exemples réussis")
```
