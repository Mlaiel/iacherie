## 📈 Exemples d'Usage

### Configuration Complète de Protection de Marque

```python
import asyncio
from backend.ai_agents.brand_agent import BrandAgentManager

async def setup_complete_brand_protection():
    # Créer un gestionnaire d'agent de marque
    manager = BrandAgentManager()
    
    # Créer un agent de marque
    brand_config = {
        "brand_name": "TechCorp",
        "industry": "technologie",
        "protection_level": "enterprise"
    }
    agent = await manager.create_agent("techcorp_brand", brand_config)
    
    # Configurer une surveillance complète
    monitoring_config = {
        "platforms": ["twitter", "facebook", "instagram", "linkedin"],
        "search_engines": ["google", "bing"],
        "marketplaces": ["amazon", "ebay"],
        "monitoring_interval": 3600,
        "alert_thresholds": {
            "sentiment_drop": 0.3,
            "mention_spike": 200,
            "crisis_keywords": ["procès", "scandale", "faux"]
        }
    }
    
    # Démarrer la surveillance
    await manager.start_monitoring("techcorp_brand", monitoring_config)
    
    return agent

# Exécuter la configuration
agent = asyncio.run(setup_complete_brand_protection())
```

### Détection de Violation en Temps Réel

```python
from backend.ai_agents.brand_agent import BrandAgent

async def detect_and_respond_violations():
    agent = BrandAgent()
    
    # Détecter les violations
    violations = await agent._detect_brand_violations({
        "search_params": {
            "brand_keywords": ["TechCorp"],
            "scope": ["marketplaces"],
            "timeframe": "24h"
        }
    })
    
    # Traiter les violations haute priorité
    for violation in violations.get("violations", []):
        if violation["threat_level"] in ["high", "critical"]:
            # Soumettre automatiquement une demande de suppression
            await agent._submit_takedown_request({
                "violation_id": violation["violation_id"],
                "platform": violation["source_platform"],
                "urgency": violation["threat_level"]
            })
            
    return violations

# Surveiller en continu
asyncio.run(detect_and_respond_violations())
```
