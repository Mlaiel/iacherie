### Echtzeit-Verletzungserkennung

```python
from backend.ai_agents.brand_agent import BrandAgent

async def detect_and_respond_violations():
    agent = BrandAgent()
    
    # Verletzungen erkennen
    violations = await agent._detect_brand_violations({
        "search_params": {
            "brand_keywords": ["TechCorp"],
            "scope": ["marketplaces"],
            "timeframe": "24h"
        }
    })
    
    # Hohe Prioritätsverletzungen verarbeiten
    for violation in violations.get("violations", []):
        if violation["threat_level"] in ["high", "critical"]:
            # Automatisch Takedown-Antrag stellen
            await agent._submit_takedown_request({
                "violation_id": violation["violation_id"],
                "platform": violation["source_platform"],
                "urgency": violation["threat_level"]
            })
            
    return violations

# Kontinuierlich überwachen
asyncio.run(detect_and_respond_violations())
```

from backend.content_protection import ContentProtectionAgent

async def integrated_protection():
    brand_agent = BrandAgent()

    # Marken-Assets mit Content-Fingerprints verknüpfen
    brand_assets = await brand_agent.get_protected_assets()
