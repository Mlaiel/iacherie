## 🚀 Getting Started

### Installation & Configuration
```bash
# Install dependencies
pip install -r requirements.txt

# Configure compliance settings
python scripts/setup_compliance.py

# Initialize legal databases
python scripts/init_legal_db.py

# Start compliance monitoring
python scripts/start_compliance_monitor.py
```

### API Integration
```python
from backend.conversational.compliance import ComplianceManager

# Initialize compliance manager
compliance = ComplianceManager()

# Validate content compliance
result = await compliance.validate_content(
    content=user_content,
    content_type="audio",
    platform="spotify",
    jurisdiction="EU"
)

# Check legal compliance
legal_status = await compliance.legal_validator.validate(
    content=user_content,
    risk_threshold=0.8
)
```


### Database Schema
```sql
-- Compliance tracking tables
CREATE TABLE compliance_sessions (
    id SERIAL PRIMARY KEY,
    session_id VARCHAR(255) NOT NULL,
    user_id INTEGER,
    compliance_score FLOAT,
    legal_warnings JSONB,
    created_at TIMESTAMP DEFAULT NOW()
);

CREATE TABLE compliance_violations (
    id SERIAL PRIMARY KEY,
    session_id VARCHAR(255),
    violation_type VARCHAR(100),
    severity VARCHAR(20),
    description TEXT,
    auto_resolved BOOLEAN DEFAULT FALSE,
    created_at TIMESTAMP DEFAULT NOW()
);
```
