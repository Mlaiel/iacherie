```python
# data_loader.py - Chargeur Haute Performance
import asyncio
from concurrent.futures import ThreadPoolExecutor
from typing import Iterator, Generator
import torch
from torch.utils.data import DataLoader, Dataset
import tensorflow as tf

class EnterpriseDataLoader:
    """Chargeur de données enterprise avec optimisations avancées"""
    
    def __init__(self, config: DatasetConfig):
        self.config = config
        self.cache_manager = CacheManager()
        self.performance_monitor = PerformanceMonitor()
        self.security_validator = SecurityValidator()
        
    async def load_dataset_async(self, 
                                dataset_path: str,
                                batch_size: int = 32,
                                num_workers: int = 8) -> DataLoader:
        """Chargement asynchrone avec optimisations"""
        
        # Validation sécurité
        await self.security_validator.validate_path(dataset_path)
        
        # Cache check
        cached_dataset = await self.cache_manager.get_cached_dataset(dataset_path)
        if cached_dataset:
            return cached_dataset
            
        # Chargement parallèle optimisé
        with ThreadPoolExecutor(max_workers=num_workers) as executor:
            dataset = await asyncio.get_event_loop().run_in_executor(
                executor, 
                self._load_dataset_sync, 
                dataset_path
            )
        
        # Optimisations performance
        dataloader = DataLoader(
            dataset,
            batch_size=batch_size,
            shuffle=True,
            num_workers=num_workers,
            pin_memory=True,
            prefetch_factor=2,
            persistent_workers=True
        )
        
        # Cache résultat
        await self.cache_manager.cache_dataset(dataset_path, dataloader)
        
        return dataloader
```

```python
        )
        
        # 4. Fusion et validation finale
        final_dataset = await self._merge_and_validate(
            refined_labels, synthetic_data, target_size
```
