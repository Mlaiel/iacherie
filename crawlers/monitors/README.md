## 📚 Usage Examples

### Basic Monitoring Setup
```python
from backend.crawlers.monitors import (
    MonitorEngine, ContentMonitor, MetricsCollector
)
from backend.crawlers.monitors.monitor_engine import MonitoringConfiguration

# Configure monitoring
config = MonitoringConfiguration(
    strategy=MonitoringStrategy.INTELLIGENT,
    mode=MonitoringMode.CONTINUOUS,
    interval=60,
    alert_thresholds={
        "cpu_usage": 80.0,
        "memory_usage": 85.0,
        "infringement_rate": 95.0
    }
)

# Initialize content monitor
content_monitor = ContentMonitor(config)
await content_monitor.start_monitoring()

# Register content for protection
content_id = await content_monitor.register_content(
    content_path="/path/to/audio.mp3",
    creator_id="user123",
    protection_level=ProtectionLevel.PREMIUM
)

# Initialize metrics collector
metrics_collector = MetricsCollector(config)
await metrics_collector.start_monitoring()

# Record custom business metrics
await metrics_collector.record_metric(
    "user_registrations_total", 
    1, 
    labels={"tier": "premium", "source": "organic"}
)
```

### Advanced Content Protection
```python
from backend.crawlers.monitors.content_monitor import (
    ContentMonitor, ProtectionLevel, InfringementSeverity
)

# Initialize protection monitor
protection_monitor = ProtectionMonitor(config)

# Add custom protection policy
await protection_monitor.add_protection_policy(
    "premium_audio_policy",
    {
        "content_types": ["audio"],
        "scan_frequency": 1800,  # 30 minutes
        "similarity_threshold": 0.90,
        "auto_takedown": True,
        "forensic_evidence": True
    }
)

# Update threat signatures
await protection_monitor.update_threat_signatures({
    "known_piracy_site": {
        "url_pattern": "piracy-site.com",
        "severity": InfringementSeverity.CRITICAL,
        "auto_block": True
    }
})

# Whitelist trusted platforms
await protection_monitor.add_to_whitelist([
    "spotify.com",
    "youtube.com",
    "soundcloud.com"
])
```

### KPI Calculation & Analytics
```python
from backend.crawlers.monitors.metrics_collector import (
    MetricsCollector, KPIDefinition, MetricCategory
)

# Define custom KPI
custom_kpi = KPIDefinition(
    name="content_engagement_score",
    description="Content engagement effectiveness",
    formula="(views + likes * 2 + shares * 3) / content_uploads_total",
    target_value=100.0,
    threshold_warning=75.0,
    threshold_critical=50.0,
    unit="score",
    category=MetricCategory.BUSINESS,
    dependencies=["views", "likes", "shares", "content_uploads_total"]
)

# Calculate KPI
metrics_collector.kpi_definitions["content_engagement_score"] = custom_kpi
kpi_result = await metrics_collector.calculate_kpi("content_engagement_score")

print(f"Engagement Score: {kpi_result.value:.2f} ({kpi_result.status})")
```
